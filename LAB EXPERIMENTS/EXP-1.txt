
import re

text = "Python is easy to learn. Python is powerful."

# Search for a pattern
result = re.search("Python", text)
if result:
    print("Pattern found:", result.group())

# Find all occurrences
matches = re.findall("Python", text)
print("All matches:", matches)

# Match at the beginning of the string
result = re.match("Python", text)
if result:
    print("Match at beginning:", result.group())

# Replace a word
new_text = re.sub("Python", "Java", text)
print("After replacement:", new_text)

OUTPUT

Pattern found: Python
All matches: ['Python', 'Python']
Match at beginning: Python
After replacement: Java is easy to learn. Java is powerful.
